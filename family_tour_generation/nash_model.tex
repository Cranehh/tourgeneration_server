\section{Decision Variables}
%% ====================================================================

For a family with $M$ members, each member $m$ has at most $T$ activities.
The per-activity decision vector is:
\begin{equation}
    \mathbf{x}_{m,t} = \bigl[\, s_{m,t},\; e_{m,t},\; p^{\mathrm{mode}}_{m,t,1}, \dots, p^{\mathrm{mode}}_{m,t,K},\; j_{m,t},\; d_{m,t} \,\bigr] \in \R^{V},
    \label{eq:decision_vector}
\end{equation}
where $V = 2 + K + 2 = 15$ (with $K=11$ travel modes), and:
\begin{itemize}
    \item $s_{m,t},\, e_{m,t}$: start and end time in z-score normalization ($\mu = 12.946$~h, $\sigma = 4.691$~h),
    \item $p^{\mathrm{mode}}_{m,t,k}$: probability of travel mode~$k$, forming a simplex $\sum_{k=1}^K p^{\mathrm{mode}}_{m,t,k} = 1$,
    \item $j_{m,t}$: joint travel probability ($\in [\varepsilon, 1-\varepsilon]$),
    \item $d_{m,t}$: driver probability ($\in [\varepsilon, 1-\varepsilon]$), where $\varepsilon = 10^{-3}$.
\end{itemize}

The full family decision vector is obtained by concatenating all per-member, per-activity vectors:
\begin{equation}
    \mathbf{x} = \bigl[\, \mathbf{x}_{1,1}, \dots, \mathbf{x}_{1,T}, \dots, \mathbf{x}_{M,1}, \dots, \mathbf{x}_{M,T} \,\bigr] \in \R^n, \quad n = M \times T \times V.
    \label{eq:full_vector}
\end{equation}

\section{Utility Function}
%% ====================================================================

The utility of member~$m$ is defined as:
\begin{equation}
\begin{split}
    U_m(\mathbf{x}) ={} & \beta_0
    + \alpha_{\mathrm{social}} \sum_{t=1}^{T} \tilde{o}_{m,t} \, a_{m,t}
    + \alpha_{\mathrm{joint}} \sum_{t=1}^{T} j_{m,t} \, a_{m,t}
    + \theta_{\mathrm{joint\_cost}} \sum_{t=1}^{T} j_{m,t}^2 \, a_{m,t} \\
    & + \theta_{\mathrm{escort}} \sum_{t=1}^{T} d_{m,t} \,(e_{m,t} - s_{m,t})\, a_{m,t} \, \indicator_{\mathrm{adult}}(m) \\
    & + \sum_{t=1}^{T} \tilde{o}_{m,t} \left( \sum_{k=1}^{K} p^{\mathrm{mode}}_{m,t,k} \, \theta_{\mathrm{travel},k} \, \E[\mathrm{TT}_{m,t,k}] \right) a_{m,t},
\end{split}
    \label{eq:utility}
\end{equation}
where:
\begin{itemize}
    \item $\beta_0$: learnable baseline utility,
    \item $\alpha_{\mathrm{social}}$: learnable social (out-of-home) preference weight,
    \item $\alpha_{\mathrm{joint}}$: learnable joint travel preference weight,
    \item $\theta_{\mathrm{joint\_cost}}$: learnable quadratic joint coordination cost (typically negative),
    \item $\theta_{\mathrm{escort}}$: learnable escort burden coefficient (typically negative),
    \item $\theta_{\mathrm{travel},k}$: learnable mode-specific travel time disutility (vector of length $K$),
    \item $\tilde{o}_{m,t} = 1 - P(\mathrm{dest}_{m,t} = \mathrm{home})$: out-of-home probability from the destination head (fixed, not optimized),
    \item $\E[\mathrm{TT}_{m,t,k}]$: expected travel time for mode~$k$ at activity~$t$, derived from the travel time matrix and destination probabilities (fixed),
    \item $a_{m,t}$: activity validity mask ($1$ if activity~$t$ of member~$m$ is valid, $0$ otherwise),
    \item $\indicator_{\mathrm{adult}}(m)$: indicator for adult members.
\end{itemize}

\paragraph{Quadratic joint coordination cost.}
The quadratic term $\theta_{\mathrm{joint\_cost}} \cdot j_{m,t}^2$ prevents corner solutions in the joint travel probability.
With $\theta_{\mathrm{joint\_cost}} < 0$ (a cost) and $\alpha_{\mathrm{joint}} > 0$ (a benefit), the first-order condition for the optimal joint probability is:
\begin{equation}
    j^*_{m,t} = \frac{\alpha_{\mathrm{joint}}}{2\,|\theta_{\mathrm{joint\_cost}}|},
    \label{eq:joint_interior}
\end{equation}
yielding an interior optimum rather than a boundary solution at~$0$ or~$1$.
For example, with $\alpha_{\mathrm{joint}} = 0.1$ and $\theta_{\mathrm{joint\_cost}} = -0.5$, the unconstrained optimum is $j^* \approx 0.1$.

\paragraph{Lower bound.}
To ensure positivity for the log-utility in the Nash welfare function, a smooth lower bound is applied:
\begin{equation}
    U_m \leftarrow U_m + \softplus(1 - U_m),
    \label{eq:utility_lb}
\end{equation}
where $\softplus(z) = \log(1 + e^z)$.
This guarantees $U_m \geq 1$ smoothly without gradient discontinuity.
Utilities of invalid members are set to $\beta_0$ to avoid affecting the Nash objective.

\section{V5: Context-Aware Utility Parameters}
\label{sec:ctx_params}
%% ====================================================================

In the base model, the utility parameters $\alpha_{\mathrm{joint}}$, $\alpha_{\mathrm{social}}$, and $\theta_{\mathrm{escort}}$ are global scalars shared across all family members and activities.
V5 introduces a \textbf{UtilityParameterNet} that predicts per-activity parameter values conditioned on the family context:
\begin{equation}
    \bigl[\alpha_{\mathrm{joint}}^{\mathrm{ctx}},\; \alpha_{\mathrm{social}}^{\mathrm{ctx}},\; \theta_{\mathrm{escort}}^{\mathrm{ctx}}\bigr]_{m,t}
    = f_{\mathrm{MLP}}\!\bigl(\mathbf{f}_{\mathrm{family}},\; \mathbf{f}_{\mathrm{member},m},\; \mathbf{p}^{\mathrm{purpose}}_{m,t}\bigr),
    \label{eq:ctx_params}
\end{equation}
where $\mathbf{f}_{\mathrm{family}} \in \R^{F_f}$ and $\mathbf{f}_{\mathrm{member},m} \in \R^{F_m}$ are the family and member attribute vectors, and $\mathbf{p}^{\mathrm{purpose}}_{m,t} \in \R^{10}$ is the predicted activity purpose probability (detached from the gradient graph).

\paragraph{Network architecture.}
The MLP consists of two linear layers with a single hidden layer of dimension~64:
\begin{equation}
    \R^{F_f + F_m + 10} \xrightarrow{\text{Linear}(64)} \xrightarrow{\text{ReLU}} \xrightarrow{\text{Dropout}(0.1)} \xrightarrow{\text{Linear}(3)} \R^{3}.
    \label{eq:mlp_arch}
\end{equation}
The three outputs are transformed via activation functions that enforce the correct sign:
\begin{align}
    \alpha_{\mathrm{joint}}^{\mathrm{ctx}} &= \softplus(z_0) \times 0.5, \label{eq:ctx_joint} \\
    \alpha_{\mathrm{social}}^{\mathrm{ctx}} &= \softplus(z_1) \times 0.5, \label{eq:ctx_social} \\
    \theta_{\mathrm{escort}}^{\mathrm{ctx}} &= -\softplus(z_2), \label{eq:ctx_escort}
\end{align}
where $[z_0, z_1, z_2]$ are the raw MLP outputs.
The $\softplus$ activation ensures $\alpha_{\mathrm{joint}}^{\mathrm{ctx}}, \alpha_{\mathrm{social}}^{\mathrm{ctx}} > 0$ (preferences) and $\theta_{\mathrm{escort}}^{\mathrm{ctx}} < 0$ (burden).

\paragraph{Initialization.}
The MLP weights are initialized with small magnitude ($\times 0.01$), and the output biases are set so that the initial predictions approximate the global parameter values.
Specifically, for a target value $\alpha^*$ with activation $\softplus(b) \times 0.5 \approx \alpha^*$, the bias is initialized as $b = \log(\exp(2\alpha^*) - 1)$.
This ensures the model starts near the baseline behavior and smoothly deviates as training progresses.

\paragraph{Integration with utility function.}
When context-aware parameters are enabled, the scalar parameters in Eq.~\eqref{eq:utility} are replaced by their per-activity counterparts:
$\alpha_{\mathrm{social}} \to \alpha_{\mathrm{social},m,t}^{\mathrm{ctx}}$,
$\alpha_{\mathrm{joint}} \to \alpha_{\mathrm{joint},m,t}^{\mathrm{ctx}}$,
$\theta_{\mathrm{escort}} \to \theta_{\mathrm{escort},m,t}^{\mathrm{ctx}}$.
The global scalar parameters remain in the model and serve as the initial reference point.

\paragraph{Gradient flow.}
During Phase~1 (SQP), the context-aware parameters are detached from the computation graph to avoid accumulating gradients through the non-differentiable SQP iterations.
Gradients flow through the UtilityParameterNet only in Phase~2 (OptNet), via the $Q_{\mathrm{diag}}$ term in the diagonal KKT backward (Eq.~\ref{eq:grad_Q}):
\begin{equation}
    \frac{\partial \mathcal{L}}{\partial \phi_{\mathrm{MLP}}}
    = \frac{\partial \mathcal{L}}{\partial \mathbf{d}^*} \cdot \frac{\partial \mathbf{d}^*}{\partial Q_{\mathrm{diag}}} \cdot \frac{\partial Q_{\mathrm{diag}}}{\partial U_m^2} \cdot \frac{\partial U_m}{\partial \alpha^{\mathrm{ctx}}} \cdot \frac{\partial \alpha^{\mathrm{ctx}}}{\partial \phi_{\mathrm{MLP}}},
    \label{eq:ctx_grad}
\end{equation}
where $\phi_{\mathrm{MLP}}$ denotes the MLP parameters.

\section{V5: Dynamic Disagreement Point}
\label{sec:dynamic_baseline}
%% ====================================================================

In the base model, the disagreement point (fallback utility when no agreement is reached) is a learnable scalar $\beta_0$ shared by all members.
V5 replaces this with a \textbf{dynamic, per-member disagreement point} that reflects each member's utility under the decoder's original (uncoordinated) prediction:
\begin{equation}
    U^{\mathrm{base}}_m = U_m\!\bigl(\thetapacked\bigr), \quad
    \beta_{0,m} = \max\!\bigl(U^{\mathrm{base}}_m,\; 1.0\bigr),
    \label{eq:dynamic_baseline}
\end{equation}
where $\thetapacked$ is the decoder output packed into decision-variable format, and $U_m(\cdot)$ is the utility function (Eq.~\ref{eq:utility}).
The $\max(\cdot, 1.0)$ clamp ensures the baseline utility is strictly positive, which is necessary for the log-utility in the Nash welfare function.

\paragraph{Economic interpretation.}
In Nash Bargaining theory, the disagreement point represents the outside option---the utility each player receives if negotiation fails.
Setting it to $U_m(\thetapacked)$ means the ``no coordination'' outcome corresponds to the decoder's independent prediction for each member.
The Nash Bargaining solution then only adjusts activities when doing so improves \emph{all} members above their uncoordinated baselines.

\paragraph{Integration.}
When dynamic baseline is enabled, the shared scalar $\beta_0$ in Eq.~\eqref{eq:utility} is replaced by the per-member value $\beta_{0,m}$, so the utility becomes:
\begin{equation}
    U_m(\mathbf{x}) = \beta_{0,m} + \alpha_{\mathrm{social}} \sum_t (\cdots) + \cdots
    \label{eq:utility_dynamic}
\end{equation}
The Nash welfare objective retains the same form $W(\mathbf{x}) = \sum_{m=1}^{M} \log U_m(\mathbf{x})$, but the member-specific baseline $\beta_{0,m}$ changes the utility landscape: members with higher uncoordinated utility (larger $\beta_{0,m}$) have a higher floor, giving them more bargaining power in the Nash solution.
The dynamic baseline is computed once (at the decoder output) and detached; it does not receive gradients.


%% ====================================================================
\section{Nash Welfare Objective}
%% ====================================================================

The Nash Bargaining solution maximizes the product of member utilities, equivalently the sum of log-utilities:
\begin{equation}
    \max_{\mathbf{x}} \; W(\mathbf{x}) = \sum_{m=1}^{M} \log U_m(\mathbf{x}).
    \label{eq:nash_welfare}
\end{equation}
The objective $W(\mathbf{x})$ is \textbf{nonlinear}: the $\log(\cdot)$ composition is inherently nonlinear, and the escort term in $U_m$ contains the bilinear product $d_{m,t}(e_{m,t} - s_{m,t})$.

\section{Phase~1: Nonlinear Program}
\label{sec:phase1}
%% ====================================================================

Phase~1 solves the full nonlinear program (NLP) to find a converged point $\xconv$.

\subsection{Optimization Problem}

\begin{equation}
    \max_{\mathbf{x}} \; W(\mathbf{x}) = \sum_{m=1}^{M} \log U_m(\mathbf{x})
    \label{eq:nlp_obj}
\end{equation}
subject to the constraints in Table~\ref{tab:phase1_constraints}.

\begin{table}[ht]
\centering
\caption{Phase~1 constraints in variable space $\mathbf{x}$.
Constraints C1--C3 are enforced in all modes (via projection and, when enabled, in the SQP subproblem).
Constraints C4--C5 are additionally included when V5 constrained SQP is enabled.}
\label{tab:phase1_constraints}
\begin{tabular}{@{}llcc@{}}
\toprule
Constraint & Formulation & Type & Mode \\
\midrule
(C1) Variable bounds &
    $x_i^{\mathrm{lb}} \leq x_i \leq x_i^{\mathrm{ub}}$ & Linear (ineq) & All \\[4pt]
(C2) Time ordering &
    $e_{m,t} \geq s_{m,t}, \;\; s_{m,t+1} \geq e_{m,t}$ & Linear (ineq) & All \\[4pt]
(C3) Mode simplex &
    $\sum_{k=1}^{K} p^{\mathrm{mode}}_{m,t,k} = 1$ & Linear (eq) & All \\[4pt]
(C4) Vehicle capacity &
    $\sum_{m,t} d_{m,t} \cdot p^{\mathrm{car}}_{m,t} \cdot \indicator_{\mathrm{adult}} \cdot a_{m,t} \leq n_{\mathrm{veh}}$ & Bilinear $\to$ linearized (ineq) & V5 \\[4pt]
(C5) Joint consistency &
    Gate-activated pairwise attribute matching & Nonlinear $\to$ gate-fixed (eq) & V5 \\
\bottomrule
\end{tabular}
\end{table}

The variable bounds (C1) are:
\begin{align}
    s_{m,t},\, e_{m,t} &\in [z_{\min}, z_{\max}], \quad z_{\min} = \frac{1 - \mu}{\sigma},\; z_{\max} = \frac{24 - \mu}{\sigma}, \\
    p^{\mathrm{mode}}_{m,t,k} &\in [\varepsilon, 1 - \varepsilon], \\
    j_{m,t},\, d_{m,t} &\in [\varepsilon, 1 - \varepsilon].
\end{align}

\paragraph{Scope of Phase~1 constraints.}
In the original mode, Phase~1 handles only the per-variable and per-activity constraints (C1--C3) via projection after each SQP step.
When V5 constrained SQP is enabled (\textsc{use\_constrained\_sqp}), all five constraints (C1--C5) are enforced within the QP subproblem at each SQP iteration (see~\S\ref{par:sqp_c4c5}).
In this mode, the vehicle capacity constraint~(C4) is linearized at the current iterate~$\mathbf{x}^{(i)}$, and the joint consistency constraint~(C5) uses a sigmoid gate evaluated and fixed at~$\mathbf{x}^{(i)}$.
In both modes, the projection operator additionally enforces vehicle and joint consistency heuristically (see Projection below).
Phase~2 (\S\ref{sec:phase2}) re-expresses all five constraints in the displacement space $\mathbf{d} = \mathbf{x} - \xconv$, linearized at the converged point~$\xconv$.

\subsection{SQP Algorithm}

Phase~1 solves the NLP using Sequential Quadratic Programming (SQP) with BFGS inverse Hessian approximation. In the original mode, constraints (C1--C3) are handled via projection after each step. In V5 constrained SQP mode, all five constraints (C1--C5) are enforced both in the QP subproblem and via projection.

\begin{algorithm}[ht]
\caption{Phase~1: SQP with BFGS and Armijo Line Search}
\label{alg:sqp}
\begin{algorithmic}[1]
\Require Decoder output $\thetapacked$, masks, utility parameters, context-aware parameters (V5, detached)
\Ensure Converged point $\xconv$

\State $\mathbf{x} \gets \thetapacked$ \Comment{Initialize at decoder output}
\State $H \gets I$ \Comment{Initial inverse Hessian approximation}

\For{$i = 0, \dots, I_{\max} - 1$}
    \State $\mathbf{g} \gets -\nabla_{\mathbf{x}} W(\mathbf{x})$ \Comment{Negative gradient of Nash welfare}
    \If{$\mathbf{g}$ contains NaN} reset $\mathbf{x} \gets \thetapacked$; \textbf{break} \EndIf
    \If{$\|\mathbf{g}\| < \epsilon_{\mathrm{tol}}$} \textbf{break} \Comment{Converged} \EndIf

    \If{$i > 0$}
        \State $\mathbf{s} \gets \mathbf{x}^{(i)} - \mathbf{x}^{(i-1)}$, \quad $\mathbf{y} \gets \mathbf{g}^{(i)} - \mathbf{g}^{(i-1)}$
        \State $H \gets \textsc{BFGS-Update}(H, \mathbf{s}, \mathbf{y})$ \Comment{With Powell's damping}
    \EndIf

    \Statex \Comment{--- Search direction (V5 branching) ---}
    \If{\textsc{use\_constrained\_sqp}} \Comment{V5: Constrained QP subproblem}
        \State $\mathbf{p} \gets \arg\min_{\mathbf{d}} \; \frac{1}{2}\mathbf{d}^\top(H{+}\delta I)\mathbf{d} + \mathbf{g}^\top\mathbf{d} \;\;\text{s.t.}\;\; G\mathbf{d}{\leq}\mathbf{h}',\; A\mathbf{d}{=}\mathbf{b}'$ \Comment{Eq.~\eqref{eq:sqp_subqp}}
        \If{QP fails or returns NaN} \Comment{Fallback}
            \State $\mathbf{p} \gets -(H + \delta I) \, \mathbf{g}$
        \EndIf
    \Else \Comment{Original: unconstrained Newton step}
        \State $\mathbf{p} \gets -(H + \delta I) \, \mathbf{g}$ \Comment{$\delta = 10^{-3}$}
    \EndIf

    \State $\alpha \gets \textsc{Armijo-Search}(\mathbf{x}, \mathbf{p})$ \Comment{Backtracking line search}
    \State $\mathbf{x} \gets \textsc{Project}(\mathbf{x} + \alpha\, \mathbf{p})$ \Comment{Project onto (C1)--(C3) + vehicle + joint}
    \If{$\mathbf{x}$ contains NaN} reset $\mathbf{x} \gets \thetapacked$; \textbf{break} \EndIf
\EndFor

\State \Return $\xconv \gets \mathbf{x}$
\end{algorithmic}
\end{algorithm}

\paragraph{V5: Constrained SQP subproblem.}
When \textsc{use\_constrained\_sqp} is enabled, the search direction at each SQP iteration is obtained by solving a constrained QP subproblem:
\begin{equation}
    \min_{\mathbf{d}} \; \frac{1}{2}\mathbf{d}^\top (H + \delta I) \mathbf{d} + \mathbf{g}^\top \mathbf{d}
    \quad\text{s.t.}\quad
    G\mathbf{d} \leq \mathbf{h}' - G\mathbf{x}, \quad
    A\mathbf{d} = \mathbf{b}' - A\mathbf{x},
    \label{eq:sqp_subqp}
\end{equation}
where $G, \mathbf{h}', A, \mathbf{b}'$ encode all five constraints (C1--C5) expressed as linear constraints on the step~$\mathbf{d}$ at the current iterate~$\mathbf{x}$:
C1 (bounds), C2 (time ordering), C3 (mode simplex), C4 (vehicle capacity, linearized), and C5 (joint consistency, gate-fixed).
The detailed formulations of C4 and C5 in the SQP subproblem are given below (\S\ref{par:sqp_c4c5}).
The inequality bounds are relaxed by $10^{-4}$ for numerical feasibility.
QR rank filtering (as in Algorithm~\ref{alg:optnet}) is applied to the equality constraints before solving.
If the QP solver fails or returns NaN/Inf values, the algorithm falls back to the unconstrained Newton direction $\mathbf{p} = -(H + \delta I)\mathbf{g}$ and relies on projection to maintain feasibility.

\paragraph{SQP subproblem constraints C4--C5.}
\label{par:sqp_c4c5}
The SQP subproblem includes all five constraints.
Constraints C1--C3 on the step~$\mathbf{d}$ take the same form as in Phase~2 (Eqs.~\ref{eq:bounds_d}--\ref{eq:mode_norm_d}), with $\xconv$ replaced by the current iterate~$\mathbf{x}^{(i)}$.
Constraints C4 and C5 are formulated as follows.

\subparagraph{(C4) Vehicle capacity (linearized at $\mathbf{x}^{(i)}$).}
The vehicle constraint involves the bilinear product $d_{m,t} \cdot p^{\mathrm{car}}_{m,t}$, where both driver probability and car mode probability are decision variables.
At the current iterate~$\mathbf{x}^{(i)}$, the constraint is linearized via first-order Taylor expansion:
\begin{equation}
    \sum_{m,t} \left( p^{\mathrm{car},(i)}_{m,t} \, d_{d_{m,t}} + d^{(i)}_{m,t} \, d_{p^{\mathrm{car}}_{m,t}} \right) \indicator_{\mathrm{adult}}(m) \, a_{m,t}
    \leq n_{\mathrm{veh}} - \sum_{m,t} d^{(i)}_{m,t} \, p^{\mathrm{car},(i)}_{m,t} \, \indicator_{\mathrm{adult}}(m) \, a_{m,t},
    \label{eq:vehicle_sqp}
\end{equation}
where superscript~$(i)$ denotes values at the current SQP iterate, $d_{d_{m,t}}$ and $d_{p^{\mathrm{car}}_{m,t}}$ are the step components for driver probability and car mode probability respectively, and $a_{m,t}$ is the activity validity mask.
This contributes a single inequality row per sample.
The key difference from the Phase~2 form (Eq.~\ref{eq:vehicle_d}) is that linearization occurs at the \emph{current} iterate~$\mathbf{x}^{(i)}$ (which changes each SQP step) rather than the fixed converged point~$\xconv$.

\subparagraph{(C5) Joint travel consistency (gate-fixed at $\mathbf{x}^{(i)}$).}
For each member pair $(m_1, m_2)$ and activity~$t$, a sigmoid gate determines whether joint consistency is enforced:
\begin{equation}
    w^{(i)}_{m_1,m_2,t} = \indicator\!\bigl[\sigma\!\bigl(10(j^{(i)}_{m_1,t} \cdot j^{(i)}_{m_2,t} - \tau)\bigr) > 0.1\bigr],
    \label{eq:joint_gate_sqp}
\end{equation}
where $\tau = 0.5$ and superscript~$(i)$ denotes evaluation at the current iterate.
For each active pair ($w^{(i)}_{m_1,m_2,t} = 1$), the step direction must drive the paired variables toward agreement:
\begin{align}
    d_{s_{m_1,t}} - d_{s_{m_2,t}} &= s^{(i)}_{m_2,t} - s^{(i)}_{m_1,t}, \label{eq:joint_start_sqp}\\
    d_{e_{m_1,t}} - d_{e_{m_2,t}} &= e^{(i)}_{m_2,t} - e^{(i)}_{m_1,t}, \label{eq:joint_end_sqp}\\
    d_{p^{\mathrm{mode}}_{m_1,t,k}} - d_{p^{\mathrm{mode}}_{m_2,t,k}} &= p^{\mathrm{mode},(i)}_{m_2,t,k} - p^{\mathrm{mode},(i)}_{m_1,t,k}, \quad \forall\, k, \label{eq:joint_mode_sqp}\\
    d_{j_{m_1,t}} - d_{j_{m_2,t}} &= j^{(i)}_{m_2,t} - j^{(i)}_{m_1,t}. \label{eq:joint_prob_sqp}
\end{align}
Each active pair contributes $2 + K + 1 = 14$ equality rows.
The right-hand sides ensure that $\mathbf{x}^{(i)} + \mathbf{d}$ yields equal attribute values for the paired members.
Unlike Phase~2 (Eqs.~\ref{eq:joint_start_d}--\ref{eq:joint_prob_d}), the gate and linearization point are re-evaluated at each SQP iteration, allowing the active set of constrained pairs to evolve as the solution progresses.

\paragraph{BFGS inverse Hessian update with Powell's damping.}
The matrix $H$ approximates $[\nabla^2(-W)]^{-1}$, updated via the damped BFGS formula.
Given step $\mathbf{s} = \mathbf{x}^{(i)} - \mathbf{x}^{(i-1)}$ and gradient change $\mathbf{y} = \mathbf{g}^{(i)} - \mathbf{g}^{(i-1)}$:
\begin{enumerate}
    \item If $\mathbf{s}^\top \mathbf{y} < 0.2\, \mathbf{s}^\top H \mathbf{s}$, apply Powell's damping:
    \begin{equation}
        \phi = \frac{0.8\, \mathbf{s}^\top H \mathbf{s}}{\mathbf{s}^\top H \mathbf{s} - \mathbf{s}^\top \mathbf{y}}, \quad
        \mathbf{y}_d = \phi\, \mathbf{y} + (1-\phi)\, H \mathbf{s}.
    \end{equation}
    Otherwise $\mathbf{y}_d = \mathbf{y}$.
    \item Apply the BFGS update:
    \begin{equation}
        H^+ = \left(I - \rho\, \mathbf{s}\, \mathbf{y}_d^\top\right) H \left(I - \rho\, \mathbf{y}_d\, \mathbf{s}^\top\right) + \rho\, \mathbf{s}\, \mathbf{s}^\top,
        \quad \rho = \frac{1}{\mathbf{s}^\top \mathbf{y}_d}.
        \label{eq:bfgs}
    \end{equation}
\end{enumerate}

\paragraph{Armijo line search.}
Starting from $\alpha = 1$, backtrack with factor $0.5$ until the sufficient decrease condition holds:
\begin{equation}
    -W\!\bigl(\textsc{Project}(\mathbf{x} + \alpha\, \mathbf{p})\bigr) < -W(\mathbf{x}) - c\, \alpha \|\mathbf{p}\|^2,
    \label{eq:armijo}
\end{equation}
where $c = 10^{-4}$ is the Armijo parameter.

\paragraph{Projection.}
The projection operator $\textsc{Project}(\cdot)$ enforces constraints (C1)--(C3) and heuristic versions of (C4)--(C5):
\begin{enumerate}
    \item \textbf{(C1)} Clamp times: $s_{m,t}, e_{m,t} \in [z_{\min}, z_{\max}]$.
    \item \textbf{(C2)} Enforce temporal ordering: for $t = 1, \dots, T$ sequentially, set $s_{m,t} \geq e_{m,t-1}$ and $e_{m,t} \geq s_{m,t}$.
    \item \textbf{(C3)} Project mode probabilities onto the simplex: clamp to $[\varepsilon, \infty)$ and renormalize.
    \item Clamp $j_{m,t}, d_{m,t} \in [\varepsilon, 1-\varepsilon]$.
    \item \textbf{(C4) Vehicle projection}: if $\sum_{m,t} d_{m,t} \cdot p^{\mathrm{car}}_{m,t} \cdot \indicator_{\mathrm{adult}}(m) \cdot a_{m,t} > n_{\mathrm{veh}}$, scale down $d_{m,t}$ proportionally so that the total driver usage equals $n_{\mathrm{veh}}$.
    \item \textbf{(C5) Joint consistency projection} (2-member families): for each active joint pair (determined by the same sigmoid gate as Eq.~\ref{eq:joint_gate_sqp}), average the start time, end time, and mode probabilities of the two members. For families with 3 or more members, joint consistency is handled exclusively by the QP equality constraints.
\end{enumerate}


%% ====================================================================
\section{Phase~2: Quadratic Program}
\label{sec:phase2}
%% ====================================================================

Phase~2 constructs a QP around the converged point $\xconv$ from Phase~1.
The displacement variable is $\mathbf{d} = \mathbf{x} - \xconv$.
The QP enforces the same five constraints (C1--C5) as V5 constrained SQP, but expressed in the displacement space $\mathbf{d}$ and linearized at the \emph{fixed} converged point~$\xconv$ (rather than the changing iterate~$\mathbf{x}^{(i)}$).

\subsection{QP Formulation}

\begin{equation}
    \min_{\mathbf{d}} \; \frac{1}{2} \mathbf{d}^\top Q \mathbf{d} + \mathbf{p}^\top \mathbf{d}
    \quad \text{s.t.} \quad
    G\mathbf{d} \leq \mathbf{h}, \quad
    A\mathbf{d} = \mathbf{b}.
    \label{eq:qp}
\end{equation}

\paragraph{Quadratic term $Q$.}
$Q$ is a diagonal matrix encoding the Nash Hessian approximation plus anchor regularization:
\begin{equation}
    Q = \mathrm{diag}\!\left(\frac{1}{U_m^2} + \lambda\right)_{i},
    \label{eq:Q}
\end{equation}
where $U_m$ is the utility of member $m$ evaluated at $\xconv$ (Eq.~\ref{eq:utility}), the index $i$ maps to member $m$ via the variable layout, and $\lambda$ is the fixed anchor weight.
The term $1/U_m^2$ approximates the diagonal of the Nash welfare Hessian $\nabla^2 W$, and $\lambda$ penalizes deviation from the decoder output $\thetapacked$.

\paragraph{Linear term $\mathbf{p}$.}
\begin{equation}
    \mathbf{p} = \lambda \left(\xconv - \thetapacked\right),
    \label{eq:p}
\end{equation}
where $\thetapacked$ is the decoder output. This term anchors the QP solution toward $\thetapacked$.

\paragraph{Nash correction weight.}
The relative influence of Nash coordination versus decoder fidelity is:
\begin{equation}
    \gamma_{\mathrm{Nash}} = \frac{1/U_m^2}{1/U_m^2 + \lambda}.
    \label{eq:correction_weight}
\end{equation}

\subsection{Inequality Constraints: $G\mathbf{d} \leq \mathbf{h}$}

\subsubsection*{Inherited from Phase~1 (re-expressed in $\mathbf{d}$)}

\paragraph{(C1$'$) Variable bounds.}
Each variable must stay within its feasible range:
\begin{equation}
    -d_i \leq x_i^{\mathrm{conv}} - x_i^{\mathrm{lb}}, \quad d_i \leq x_i^{\mathrm{ub}} - x_i^{\mathrm{conv}}.
    \label{eq:bounds_d}
\end{equation}
This yields $2n$ inequality rows.

\paragraph{(C2$'$) Time ordering.}
The temporal constraints $e_{m,t} \geq s_{m,t}$ and $s_{m,t+1} \geq e_{m,t}$ become:
\begin{align}
    d_{s_{m,t}} - d_{e_{m,t}} &\leq e^{\mathrm{conv}}_{m,t} - s^{\mathrm{conv}}_{m,t}, \label{eq:time_d1} \\
    d_{e_{m,t}} - d_{s_{m,t+1}} &\leq s^{\mathrm{conv}}_{m,t+1} - e^{\mathrm{conv}}_{m,t}. \label{eq:time_d2}
\end{align}
These remain linear, as in Phase~1.

\subsubsection*{Family-level coordination constraints}

\paragraph{(C4$'$) Vehicle capacity.}
The vehicle capacity constraint $\sum_{m,t} d_{m,t} \cdot p^{\mathrm{car}}_{m,t} \cdot \indicator_{\mathrm{adult}}(m) \cdot a_{m,t} \leq n_{\mathrm{veh}}$ involves the bilinear product $d_{m,t} \cdot p^{\mathrm{car}}_{m,t}$ (both are decision variables), which cannot be directly included in a QP.
It is linearized via first-order Taylor expansion around $\xconv$:
\begin{equation}
    \sum_{m,t} \left( p^{\mathrm{car},0}_{m,t} \, \Delta d_{m,t} + d^{0}_{m,t} \, \Delta p^{\mathrm{car}}_{m,t} \right) \indicator_{\mathrm{adult}}(m) \, a_{m,t}
    \leq n_{\mathrm{veh}} - \sum_{m,t} d^0_{m,t} \, p^{\mathrm{car},0}_{m,t} \, \indicator_{\mathrm{adult}}(m) \, a_{m,t},
    \label{eq:vehicle_d}
\end{equation}
where superscript~$0$ denotes values at $\xconv$, and $\Delta$ denotes the displacement component.
This linearization is valid locally around $\xconv$.
It contributes a single linear inequality row.
The same linearization structure is used in the SQP subproblem (Eq.~\ref{eq:vehicle_sqp}), but evaluated at the current iterate~$\mathbf{x}^{(i)}$ rather than~$\xconv$.

\subsection{Equality Constraints: $A\mathbf{d} = \mathbf{b}$}

\subsubsection*{Inherited from Phase~1 (re-expressed in $\mathbf{d}$)}

\paragraph{(C3$'$) Mode probability normalization.}
Since $\sum_k p^{\mathrm{mode}}_{m,t,k} = 1$ holds at $\xconv$, the displacement must preserve this:
\begin{equation}
    \sum_{k=1}^{K} \Delta p^{\mathrm{mode}}_{m,t,k} = 0, \quad \forall\, m, t.
    \label{eq:mode_norm_d}
\end{equation}

\subsubsection*{Joint travel coordination}

\paragraph{(C5$'$) Joint travel consistency.}
When two members $m_1, m_2$ are both engaged in joint travel at activity~$t$, their activity attributes should agree.
The activation is determined by a sigmoid gate evaluated at $\xconv$ and fixed as a binary decision:
\begin{equation}
    w_{m_1,m_2,t} = \indicator\!\bigl[\sigma\!\bigl(10(j_{m_1,t}^0 \cdot j_{m_2,t}^0 - \tau)\bigr) > 0.1\bigr],
    \label{eq:joint_gate_d}
\end{equation}
where $\tau = 0.5$ is the joint consistency threshold, $\sigma(\cdot)$ is the sigmoid function, and superscript $0$ denotes values at $\xconv$.
The sigmoid of the bilinear product $j_{m_1,t} \cdot j_{m_2,t}$ is inherently nonlinear; evaluating it at $\xconv$ and fixing the result as a binary gate converts this into a fixed set of linear equalities suitable for the QP.
The same gate structure is used in the SQP subproblem (Eq.~\ref{eq:joint_gate_sqp}), but evaluated at the current iterate~$\mathbf{x}^{(i)}$.

For each active pair ($w_{m_1,m_2,t} = 1$):
\begin{align}
    \Delta s_{m_1,t} - \Delta s_{m_2,t} &= s^0_{m_2,t} - s^0_{m_1,t}, \label{eq:joint_start_d}\\
    \Delta e_{m_1,t} - \Delta e_{m_2,t} &= e^0_{m_2,t} - e^0_{m_1,t}, \label{eq:joint_end_d}\\
    \Delta p^{\mathrm{mode}}_{m_1,t,k} - \Delta p^{\mathrm{mode}}_{m_2,t,k} &= p^{\mathrm{mode},0}_{m_2,t,k} - p^{\mathrm{mode},0}_{m_1,t,k}, \quad \forall\, k, \label{eq:joint_mode_d}\\
    \Delta j_{m_1,t} - \Delta j_{m_2,t} &= j^0_{m_2,t} - j^0_{m_1,t}. \label{eq:joint_prob_d}
\end{align}
Each active pair contributes $2 + K + 1 = 14$ equality rows.
Note that the right-hand sides drive the paired variables toward agreement: if the pair is active and the values differ at $\xconv$, the QP will force them to converge.

\subsection{QP Solve}

\begin{algorithm}[ht]
\caption{Phase~2: QP Solve with Gradient Re-attachment}
\label{alg:optnet}
\begin{algorithmic}[1]
\Require $\xconv$, $\thetapacked$ (decoder output), masks, utility parameters, V5: context-aware parameters, dynamic baseline
\Ensure $\xfinal$ (with gradient to decoder, utility parameters, and UtilityParameterNet)

\Statex \textbf{--- Step 1: Construct QP parameters ---}
\State $U_m \gets$ Eq.~\eqref{eq:utility} at $\xconv$ \Comment{V5: uses ctx\_params and dynamic baseline}
\State $Q_{\mathrm{diag}} \gets 1/U_m^2 + \lambda$ \quad (expanded per variable via member mapping) \Comment{Eq.~\eqref{eq:Q}}
\State $\mathbf{p} \gets \lambda(\xconv - \thetapacked)$ \Comment{Eq.~\eqref{eq:p}}

\Statex \textbf{--- Step 2: Build constraints (C1$'$--C5) ---}
\State $(G, \mathbf{h}) \gets$ Eqs.~\eqref{eq:bounds_d}--\eqref{eq:vehicle_d} \Comment{C1$'$, C2$'$, C4}
\State $(A, \mathbf{b}) \gets$ Eqs.~\eqref{eq:mode_norm_d}--\eqref{eq:joint_prob_d} \Comment{C3$'$, C5}

\Statex \textbf{--- Step 2.5: Constraint preprocessing ---}
\State Remove all-zero rows from $A$ and $G$ \Comment{Inactive constraints for masked variables}
\State $A \gets \textsc{QR-Rank-Filter}(A)$ \Comment{Eq.~\eqref{eq:qr_filter}; remove linearly dependent rows}
\State $\mathbf{h} \gets \mathbf{h} + 10^{-4}$ \Comment{Slight relaxation for numerical feasibility}

\Statex \textbf{--- Step 3: Solve QP (under \texttt{no\_grad}) ---}
\State $\mathbf{d}^* \gets \arg\min_{\mathbf{d}} \; \tfrac{1}{2} \mathbf{d}^\top Q \mathbf{d} + \mathbf{p}^\top \mathbf{d} \;\;\text{s.t.}\;\; G\mathbf{d} \leq \mathbf{h},\; A\mathbf{d} = \mathbf{b}$
\If{$\mathbf{d}^*$ contains NaN or Inf} $\mathbf{d}^* \gets \mathbf{0}$ \EndIf

\Statex \textbf{--- Step 4: Re-attach gradient ---}
\State $\mathbf{d}^* \gets \textsc{DiagonalKKTBackward}(\mathbf{p}, Q_{\mathrm{diag}}, \mathbf{d}^*)$ \Comment{See \S\ref{sec:kkt_backward}}

\Statex \textbf{--- Step 5: Assemble final output ---}
\State $\xfinal \gets \thetapacked + (\xconv - \thetapacked)_{\mathrm{detach}} + \mathbf{d}^*$ \Comment{Eq.~\eqref{eq:x_final}}
\State $\xfinal \gets \textsc{Project}(\xfinal)$
\If{$\xfinal$ contains NaN} $\xfinal \gets \thetapacked$ \Comment{Full fallback} \EndIf

\State \Return $\xfinal$
\end{algorithmic}
\end{algorithm}

\paragraph{QR rank filtering.}
Before passing the equality constraint matrix $A$ to the QP solver, linearly dependent rows are removed to prevent singular KKT systems.
This is achieved via QR decomposition:
\begin{equation}
    A^\top = QR, \quad \text{keep row } i \text{ if } |R_{ii}| > 10^{-8}.
    \label{eq:qr_filter}
\end{equation}
The filtering uses the first batch element (constraint structure is similar across the batch) and is applied to both Phase~1 constrained SQP subproblems and the Phase~2 QP.


\subsection{Final Solution Assembly}

The final output is composed as:
\begin{equation}
    \xfinal = \thetapacked + \underbrace{(\xconv - \thetapacked)}_{\text{no gradient}} + \underbrace{\mathbf{d}^*(\thetapacked, Q)}_{\text{carries gradient}},
    \label{eq:x_final}
\end{equation}
where the SQP displacement $(\xconv - \thetapacked)$ is treated as a constant (no gradient flows through Phase~1), and $\mathbf{d}^*$ depends on $\thetapacked$ through $\mathbf{p}$ (Eq.~\ref{eq:p}) and on utility parameters through $Q$ (Eq.~\ref{eq:Q}).


\subsection{Diagonal KKT Backward}
\label{sec:kkt_backward}

For the diagonal QP $\min_{\mathbf{d}} \frac{1}{2}\mathbf{d}^\top \mathrm{diag}(Q_{\mathrm{diag}})\, \mathbf{d} + \mathbf{p}^\top \mathbf{d}$, the unconstrained closed-form solution is $\mathbf{d}^* = -\mathbf{p} / Q_{\mathrm{diag}}$.
We use this diagonal structure to approximate the gradients of the constrained solution:
\begin{equation}
    \frac{\partial d^*_i}{\partial p_i} \approx -\frac{1}{Q_{\mathrm{diag},i}}, \qquad
    \frac{\partial d^*_i}{\partial Q_{\mathrm{diag},i}} \approx -\frac{d^*_i}{Q_{\mathrm{diag},i}}.
    \label{eq:grad_reattach}
\end{equation}

Given upstream gradient $\bar{\mathbf{d}}$ from the loss function, the gradients propagate as:
\begin{align}
    \bar{\mathbf{p}} &= -\bar{\mathbf{d}} \,/\, Q_{\mathrm{diag}} & &\longrightarrow \text{gradient to } \thetapacked \text{ (decoder)}, \label{eq:grad_p} \\
    \bar{Q}_{\mathrm{diag}} &= -\bar{\mathbf{d}} \odot \mathbf{d}^* \,/\, Q_{\mathrm{diag}} & &\longrightarrow \text{gradient to utility parameters via } 1/U_m^2. \label{eq:grad_Q}
\end{align}


%% ====================================================================
\section{Straight-Through Log Unpack}
\label{sec:st_log}
%% ====================================================================

After optimization, probability variables ($p^{\mathrm{mode}}, j, d$) must be converted back to log-probabilities for the downstream cross-entropy loss.
A naive $\log(p)$ introduces a $1/p$ factor in the gradient, causing instability for small probabilities.

The straight-through log estimator is applied:
\begin{equation}
    \tilde{p}_k = p_k + \bigl(\log p_k - p_k\bigr)_{\mathrm{detach}},
    \label{eq:st_log}
\end{equation}
so that the forward value equals $\log p_k$ while the backward gradient is the identity ($\partial \tilde{p}_k / \partial p_k = 1$).
For binary heads (joint, driver), the scalar $p$ is expanded to $[1-p,\; p]$ before applying Eq.~\eqref{eq:st_log}.

\section{Two-Phase Flow Summary}
\label{sec:flow}
%% ====================================================================

\begin{enumerate}
    \item \textbf{Pack}: Convert decoder predictions (logits) to probability-space vector $\thetapacked$.
    \item \textbf{Phase~1 (SQP)}: Starting from $\mathbf{x}_0 = \thetapacked$, iterate BFGS + Armijo to find $\xconv$ (Alg.~\ref{alg:sqp}). Handles per-variable constraints (C1--C3) via projection.
    \item \textbf{Phase~2 (QP)}: Construct QP (Eq.~\ref{eq:qp}) using $\xconv$ and $\thetapacked$. Enforces all constraints: (C1$'$--C3$'$) inherited from Phase~1, plus \textbf{new} family-level coordination constraints (C4) vehicle capacity and (C5) joint consistency, linearized at $\xconv$. Gradient flows via diagonal KKT backward (Alg.~\ref{alg:optnet}).
    \item \textbf{Project}: Enforce feasibility on $\xfinal$.
    \item \textbf{Unpack}: Convert back to prediction dict using straight-through log (Eq.~\ref{eq:st_log}).
\end{enumerate}

Gradient flows from loss $\to$ unpack $\to$ $\xfinal$ $\to$ $\mathbf{d}^*$ $\to$
\begin{itemize}
    \item $\mathbf{p} \to \thetapacked \to$ decoder parameters,
    \item $Q_{\mathrm{diag}} \to 1/U_m^2 \to U_m \to$ learnable utility parameters ($\alpha_{\mathrm{joint}}, \alpha_{\mathrm{social}}, \theta_{\mathrm{escort}}, \theta_{\mathrm{joint\_cost}}, \beta_0, \theta_{\mathrm{travel}}$).
\end{itemize}

\paragraph{V5 additional gradient paths.}
When context-aware parameters and dynamic baseline are enabled, two additional paths emerge:
\begin{itemize}
    \item $Q_{\mathrm{diag}} \to 1/U_m^2 \to U_m \to \alpha^{\mathrm{ctx}} \to \phi_{\mathrm{MLP}}$ (UtilityParameterNet parameters, Eq.~\ref{eq:ctx_grad}),
    \item $Q_{\mathrm{diag}} \to 1/U_m^2 \to U_m \to \theta_{\mathrm{joint\_cost}}$ (quadratic joint cost parameter).
\end{itemize}
Note that context-aware parameters are \emph{detached} during Phase~1 (SQP); their gradients flow exclusively through Phase~2 (OptNet).
The dynamic baseline $\beta_{0,m}$ is also detached (computed once at the decoder output) and does not receive gradients directly, but it influences the $Q_{\mathrm{diag}}$ values and thereby the optimization landscape.
